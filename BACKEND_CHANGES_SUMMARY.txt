def format_ai_instructions(blocks):
    result = []
    for block in blocks:
        if 'ai_instructions' in block:
            result.append(f"""
            {block['type']}:
            - Guide: {block['ai_instructions']['generation_guide']}
            - Rules: {', '.join(block['ai_instructions']['quality_rules'])}
            """)
    return '\n'.join(result)
